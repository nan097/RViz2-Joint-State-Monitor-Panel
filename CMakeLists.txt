cmake_minimum_required(VERSION 3.16)
project(joint_state_rviz_panel)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(yaml-cpp REQUIRED)
# 数据缓存与配置解析独立于 RViz 界面，便于单元测试。
add_library(monitor_core STATIC src/joint_data_manager.cpp src/robot_config.cpp)
set_target_properties(monitor_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(monitor_core PUBLIC include)
ament_target_dependencies(monitor_core sensor_msgs)
target_link_libraries(monitor_core yaml-cpp)
# 产品仅生成 RViz 面板共享库，不创建独立监视进程。
add_library(${PROJECT_NAME} SHARED src/joint_state_panel.cpp src/joint_row_widget.cpp
  include/joint_state_rviz_panel/joint_state_panel.hpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} rclcpp sensor_msgs rviz_common pluginlib)
target_link_libraries(${PROJECT_NAME} monitor_core Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY config rviz DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS test/mock_joint_states_publisher.py DESTINATION lib/${PROJECT_NAME})
# 测试宿主只用于验证，不安装到产品运行目录。
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  add_executable(rviz_host_smoke test/rviz_host_smoke.cpp)
  ament_target_dependencies(rviz_host_smoke rclcpp sensor_msgs rviz_common)
  target_link_libraries(rviz_host_smoke ${PROJECT_NAME} Qt5::Widgets)
  ament_add_gtest(test_monitor test/test_monitor.cpp)
  target_link_libraries(test_monitor monitor_core)
  ament_add_gtest(test_panel test/test_panel.cpp)
  ament_target_dependencies(test_panel rclcpp sensor_msgs rviz_common pluginlib)
  target_link_libraries(test_panel ${PROJECT_NAME} Qt5::Widgets)
  set_tests_properties(test_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_LOG_DIR=${CMAKE_CURRENT_BINARY_DIR}/ros_logs" TIMEOUT 60)
endif()
ament_package()
